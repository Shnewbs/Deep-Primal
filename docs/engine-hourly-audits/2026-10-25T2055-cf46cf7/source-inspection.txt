6: /** Payload-byte budget plus an entry ceiling. Callers retain ownership of returned values. */
7: public final class ByteLru<K,V> {
8:     private final LinkedHashMap<K,V> entries = new LinkedHashMap<>(16, .75f, true);
9:     private final long limit;
10:     private final int maximum;
11:     private final ToLongFunction<V> weigh;
12:     private long bytes, hits;
13:     public ByteLru(long limit, int maximum, ToLongFunction<V> weigh) {
14:         if (limit < 0 || maximum < 0) throw new IllegalArgumentException();
15:         this.limit = limit; this.maximum = maximum; this.weigh = weigh;
16:     }
17:     public V get(K key) { V value = entries.get(key); if (value != null) hits++; return value; }
18:     public void put(K key, V value) {
19:         long size = weigh.applyAsLong(value);
20:         if (size < 0) throw new IllegalArgumentException("Negative payload size");
21:         V old = entries.remove(key); if (old != null) bytes -= weigh.applyAsLong(old);
22:         if (size > limit || maximum == 0) return;
23:         entries.put(key, value); bytes += size;
24:         while (bytes > limit || entries.size() > maximum) {
25:             var it = entries.entrySet().iterator(); bytes -= weigh.applyAsLong(it.next().getValue()); it.remove();
26:         }
27:     }
28:     public long bytes() { return bytes; }
29:     public long hits() { return hits; }
30:     public int size() { return entries.size(); }
31:     public void clear() { entries.clear(); bytes = hits = 0; }
32: }
