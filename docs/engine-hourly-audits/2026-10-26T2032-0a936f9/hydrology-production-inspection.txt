419:   static int streamAllowance(int quota, long start, long now) {
420:     if (start == Long.MIN_VALUE || now < start) return quota;
421:     // With ordered signed ticks, negative subtraction means elapsed-time overflow.
422:     // Such an interval is already complete; never turn it into a negative quota.
423:     long elapsed = now - start;
424:     long phase = elapsed < 0 ? 19 : Math.min(19, elapsed);
425:     return (int) (quota * (phase + 1) / 20);
426:   }
