12:/**
13: * World generation pipeline coordinating terrain, water, and biome generation.
14: * Designed for parallel execution with proper dependency ordering.
15: */
16:public final class WorldGenPipeline {
17:    private final TerrainEngine terrainEngine;
18:    private final ChunkMaterializer materializer;
19:    private final ChunkCache densityCache;
20:    private final ChunkCache meshCache;
21:    private final Executor generationExecutor;
22:    private final Executor callbackExecutor;
23:    
24:    // Pipeline stages
25:    private final Stage<ChunkKey, DensityGraph> densityStage;
26:    private final Stage<DensityGraph, ChunkMaterializer.MeshData> meshStage;
27:    private final Stage<ChunkMaterializer.MeshData, ChunkMaterializer.MaterializedChunk> gpuStage;
28:
29:    public WorldGenPipeline(TerrainEngine terrainEngine, ChunkMaterializer materializer,
30:                            ChunkCache densityCache, ChunkCache meshCache) {
31:        this.terrainEngine = terrainEngine;
32:        this.materializer = materializer;
33:        this.densityCache = densityCache;
34:        this.meshCache = meshCache;
35:        
36:        // Dedicated executor for world generation
37:        this.generationExecutor = Executors.newFixedThreadPool(
38:            Runtime.getRuntime().availableProcessors(),
39:            new ThreadFactory() {
40:                private int count = 0;
41:                @Override
42:                public Thread newThread(Runnable r) {
43:                    Thread t = new Thread(r, "DeepPrimal-WorldGen-" + count++);
44:                    t.setDaemon(true);
45:                    t.setPriority(Thread.NORM_PRIORITY - 1);
46:                    return t;
47:                }
48:            }
49:        );
50:        
51:        // Callback executor (can be main thread)
52:        this.callbackExecutor = Executors.newSingleThreadExecutor(r -> {
53:            Thread t = new Thread(r, "DeepPrimal-WorldGen-Callback");
54:            t.setDaemon(true);
55:            return t;
56:        });
57:        
58:        // Build pipeline stages
59:        this.densityStage = new Stage<>("density", generationExecutor, this::generateDensity);
60:        this.meshStage = new Stage<>("mesh", generationExecutor, this::buildMesh);
61:        this.gpuStage = new Stage<>("gpu-upload", callbackExecutor, this::uploadToGpu);
62:    }
63:
64:    /**
65:     * Generate a chunk through the full pipeline.
66:     */
67:    public CompletableFuture<ChunkMaterializer.MaterializedChunk> generateChunk(ChunkKey key) {
68:        return densityStage.process(key)
69:            .thenCompose(density -> meshStage.process(density))
70:            .thenCompose(mesh -> gpuStage.process(mesh));
71:    }
72:
73:    /**
74:     * Generate multiple chunks in parallel.
75:     */
76:    public CompletableFuture<Void> generateChunks(Iterable<ChunkKey> keys, Consumer<ChunkMaterializer.MaterializedChunk> onComplete) {
77:        var futures = new java.util.ArrayList<CompletableFuture<ChunkMaterializer.MaterializedChunk>>();
78:        for (ChunkKey key : keys) {
79:            futures.add(generateChunk(key).thenAccept(onComplete));
80:        }
81:        return CompletableFuture.allOf(futures.toArray(new CompletableFuture[0]));
82:    }

84:    private DensityGraph generateDensity(ChunkKey key) {
85:        return densityCache.getOrCompute(key.asLong(), k -> {
86:            long start = System.nanoTime();
87:            try {
88:                DensityField field = new DensityField(terrainEngine.noise, terrainEngine.options);
89:                return field.sampleChunk(key.chunkX() * 16, key.chunkZ() * 16);
90:            } finally {
91:                EngineMetrics.recordTerrainGenTime(System.nanoTime() - start);
92:            }
93:        }, estimateDensitySize());
