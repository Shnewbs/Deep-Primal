21:    */

23:       boolean focused, int fpsLimit, int fullDistance, int verticalDistance, int lodDistance) {
24:     boolean changed = !initialized || this.world != world || this.width != width
25:         || this.height != height || this.focused != focused || this.fpsLimit != fpsLimit
26:         || this.fullDistance != fullDistance || this.verticalDistance != verticalDistance
27:         || this.lodDistance != lodDistance || now - previous < 0;
28:     if (changed) {

30:       // This includes all intervening waits; it is not an isolated world-load timer.
31:       if (initialized && now - previous >= 0) {
32:         transitions.record(now - previous);
33:         transitionCount++;
34:       }
35:       intervals.reset();
36:       generation++;
37:       this.world = world;
38:       this.width = width;
39:       this.height = height;

42:       this.fullDistance = fullDistance;
43:       this.verticalDistance = verticalDistance;
44:       this.lodDistance = lodDistance;

47:     previous = now;
48:     return changed;
49:   }
