51:         self.seq = 0

58:     def read(self, count, deadline):
59:         data = b''
60:         while len(data) < count:
61:             remaining = deadline - time.monotonic()
62:             if remaining <= 0:
63:                 raise TimeoutError('RCON request deadline exceeded')
64:             self.sock.settimeout(remaining)
65:             part = self.sock.recv(count - len(data))
66:             if not part:
67:                 raise RuntimeError('RCON disconnected')
68:             data += part
69:         return data

71:     def request(self, kind, text):
72:         deadline = time.monotonic() + self.REQUEST_SECONDS
73:         self.seq += 1
74:         body = struct.pack('<ii', self.seq, kind) + text.encode() + b'\0\0'
75:         self.sock.settimeout(self.REQUEST_SECONDS)
76:         self.sock.sendall(struct.pack('<i', len(body)) + body)
77:         # Some servers send one empty RESPONSE_VALUE before AUTH_RESPONSE.
78:         # Never accept that preamble as authentication or skip arbitrary packets.
79:         for index in range(2 if kind == 3 else 1):
80:             length = struct.unpack('<i', self.read(4, deadline))[0]
81:             if not 10 <= length <= self.MAX_RESPONSE_BYTES:
82:                 raise RuntimeError('Invalid RCON response length: ' + str(length))
83:             response = self.read(length, deadline)
84:             request_id, response_kind = struct.unpack('<ii', response[:8])
85:             if request_id == -1:
86:                 raise RuntimeError('RCON authentication failed')
87:             if request_id != self.seq or response[-2:] != b'\0\0':
88:                 raise RuntimeError('Invalid RCON response identity or terminators')
89:             if kind == 3 and index == 0 and response_kind == 0 and length == 10:
90:                 continue
91:             if response_kind != (2 if kind == 3 else 0):
92:                 raise RuntimeError('Invalid RCON response type')
93:             return response[8:-2].decode()
94:         raise RuntimeError('Missing RCON authentication response')
