19:  }

21:    Entry e=entries.get(Objects.requireNonNull(key));
22:    if(e==null){misses.incrementAndGet();EngineMetrics.CACHE_MISSES.increment();return null;}
23:    hits.incrementAndGet();EngineMetrics.CACHE_HITS.increment();return (T)e.value;
24:  }
25:  public <T> T getOrCompute(Object key,Function<Object,T> computer,long weight) {
26:    Objects.requireNonNull(key);Objects.requireNonNull(computer);
27:    if(weight<0)throw new IllegalArgumentException("negative cache weight");
28:    final long observed;
29:    synchronized(this){T ready=get(key);if(ready!=null)return ready;observed=generation;}
30:    T computed=Objects.requireNonNull(computer.apply(key));
31:    synchronized(this){
32:      // clear() invalidates pending admissions without waiting for terrain computation.
33:      if(generation!=observed)return computed;
34:      Entry existing=entries.get(key);
35:      if(existing!=null){@SuppressWarnings("unchecked") T winner=(T)existing.value;return winner;}
36:      put(key,computed,weight);
37:      return computed;
38:    }
39:  }

41:    Objects.requireNonNull(key);Objects.requireNonNull(value);
42:    if(weight<0)throw new IllegalArgumentException("negative cache weight");
43:    remove(key);
44:    if(weight>capacity || capacity==0)return;
45:    while(!entries.isEmpty() && (bytes>capacity-weight || entries.size()>=limit))evict();
46:    entries.put(key,new Entry(value,weight));bytes+=weight;
47:  }
48:  private void evict(){var it=entries.entrySet().iterator();bytes-=it.next().getValue().bytes;it.remove();evictions.incrementAndGet();EngineMetrics.CACHE_EVICTIONS.increment();}

7:/** Coalesced, bounded CPU queries. Worlds and dimensions are part of cache identity. */
8:public final class WorldGenPipeline {
9:  private final ChunkMaterializer materializer;private final ChunkCache cache;
10:  private final Map<ChunkKey,CompletableFuture<MaterializedChunk>> pending=new HashMap<>();
11:  private volatile boolean stopped;
12:  public WorldGenPipeline(ChunkMaterializer materializer,ChunkCache cache){this.materializer=materializer;this.cache=cache;}
13:  public synchronized CompletableFuture<MaterializedChunk> generateChunk(ChunkKey key){
14:    if(stopped)return CompletableFuture.failedFuture(new IllegalStateException("pipeline stopped"));
15:    MaterializedChunk ready=cache.get(key);if(ready!=null)return CompletableFuture.completedFuture(ready);
16:    var existing=pending.get(key);if(existing!=null)return existing.copy();
17:    if(pending.size()>=64)return CompletableFuture.failedFuture(new RejectedExecutionException("64 pending chunk queries"));
18:    var result=new CompletableFuture<MaterializedChunk>();pending.put(key,result);
19:    try{materializer.materializeAsync(key,()->stopped).whenComplete((chunk,error)->{
20:      Throwable failure=error;
21:      synchronized(this){
22:        pending.remove(key);
23:        if(stopped)failure=new CancellationException("pipeline stopped");
24:        else if(failure==null)cache.put(key,chunk,chunk.payloadBytes());
25:      }
26:      // User continuations may call back into EngineCore: never run them under this lock.
27:      if(failure!=null)result.completeExceptionally(failure);else result.complete(chunk);
28:    });}catch(RuntimeException error){pending.remove(key);result.completeExceptionally(error);}
29:    return result.copy();
30:  }
31:  public CompletableFuture<Void> generateChunks(Iterable<ChunkKey> keys,Consumer<MaterializedChunk> consumer){
32:    Objects.requireNonNull(consumer);
33:    var batch=new ArrayList<ChunkKey>();for(var key:keys){if(batch.size()>=4096)return CompletableFuture.failedFuture(new IllegalArgumentException("batch exceeds 4096"));batch.add(Objects.requireNonNull(key));}
34:    Iterator<ChunkKey> iterator=batch.iterator();
35:    CompletableFuture<Void> chain=CompletableFuture.completedFuture(null);
36:    // At most 4096 items per explicit batch; sequential admission avoids queue floods.
37:    int count=0;while(iterator.hasNext()){if(++count>4096)return CompletableFuture.failedFuture(new IllegalArgumentException("batch exceeds 4096"));ChunkKey key=iterator.next();chain=chain.thenCompose(v->generateChunk(key).thenAccept(consumer));}return chain;
38:  }
39:  public void shutdown(){
40:    List<CompletableFuture<MaterializedChunk>> cancelled;
41:    synchronized(this){stopped=true;cancelled=new ArrayList<>(pending.values());pending.clear();cache.clear();}
42:    for(var future:cancelled)future.completeExceptionally(new CancellationException("pipeline stopped"));
43:  }
44:  public synchronized int pendingCount(){return pending.size();}
