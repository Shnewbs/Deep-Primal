3: /**
4:  * Presentation only. Retarget from the currently displayed value, never from the last tick.
5:  * Versioned (surface height) samples settle finite-parcel ping-pong: a change of at most one
6:  * unit that reverts within {@link #SETTLE_MILLIS} shows the pair's midpoint, so steady through-flow
7:  * renders as one still surface instead of bobbing 1/16 block per passing parcel. Unchanged
8:  * recaptures do not extend that hold; a cleanup retarget returns stopped flow to its real height.
9:  */
10: public final class WaterSurfaceTransition {
11:   public static final int MILLIS = 100;
12:   public static final int SETTLE_MILLIS = 400;
13:   public static final float SETTLE_BAND = 1 / 16f + 0.0001f;
14:   private float from, target;
15:   private float lastRaw, previousRaw = Float.NaN;
16:   private long rawAt = Long.MIN_VALUE / 2;
17:   private long settleUntil = Long.MIN_VALUE;
18:   private long started;
19:   private long revision = Long.MIN_VALUE;
20: 
21:   public WaterSurfaceTransition(float initial, long now) {
22:     from = target = lastRaw = initial;
23:     started = now;
24:   }
25: 
26:   public float at(long now) {
27:     float t = Math.clamp((now - started) / (float) MILLIS, 0, 1);
28:     return from + (target - from) * t;
29:   }
30: 
31:   public void target(float next, long now) {
32:     if (Math.abs(next - target) < 0.00001f) return;
33:     from = at(now);
34:     target = next;
35:     started = now;
36:   }
37: 
38:   public void target(float next, long now, long version) {
39:     if (version < revision) return;
40:     revision = version;
41:     advance(now);
42:     // Neighboring pages resubmit the same shared corner; that is not a new sample.
43:     if (Math.abs(next - lastRaw) < 0.00001f) return;
44:     boolean reverts = Math.abs(next - previousRaw) < 0.00001f
45:         && Math.abs(next - lastRaw) <= SETTLE_BAND
46:         && now - rawAt <= SETTLE_MILLIS;
47:     float shown = reverts ? (next + lastRaw) * .5f : next;
48:     settleUntil = reverts ? now + SETTLE_MILLIS : Long.MIN_VALUE;
49:     previousRaw = lastRaw;
50:     lastRaw = next;
51:     rawAt = now;
52:     target(shown, now);
53:   }
54: 
55:   /** Service presentation expiry even when the page that last changed is no longer visible. */
56:   public void advance(long now) {
57:     if (settleUntil != Long.MIN_VALUE && now >= settleUntil) {
58:       settleUntil = Long.MIN_VALUE;
59:       target(lastRaw, now);
60:     }
61:   }

64:     return target;
65:   }
66:

68:   public int remaining(long now) {
69:     long animation = Math.abs(from - target) < .00001f ? 0 : Math.max(0, MILLIS - (now - started));
70:     return (int) Math.clamp(Math.max(animation, settleUntil == Long.MIN_VALUE ? 0 : settleUntil - now),
71:         0, Integer.MAX_VALUE);
72:   }

76:     started = now;
77:   }
