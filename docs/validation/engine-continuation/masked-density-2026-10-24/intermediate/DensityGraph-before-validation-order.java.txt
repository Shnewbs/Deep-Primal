package com.shnewbs.deepprimal.core.density;

import java.util.*;

/** Versioned, immutable density program. Positive output means solid for terrain consumers. */
public final class DensityGraph implements DensityField {
  public static final int VERSION = 1, MAX_NODES = 128, MAX_DEPTH = 24, MAX_COST = 4096;
  public static final int WORLD_LIMIT = 30_000_000;
  private static final double VALUE_LIMIT = 1e12;

  public enum Op {
    CONSTANT(0,1), X(0,0), Y(0,0), Z(0,0), NOISE_2D(0,0), NOISE_3D(0,0),
    ADD(2,0), MULTIPLY(2,0), MIN(2,0), MAX(2,0), ABS(1,0), CLAMP(1,2),
    SMOOTHSTEP(1,2), LERP(3,0), WARP(4,1);
    final int inputs, parameters;
    Op(int inputs,int parameters) { this.inputs=inputs; this.parameters=parameters; }
  }

  public record Node(String id, Op op, List<String> inputs, List<Double> parameters, NoiseSpec noise) {
    public Node {
      if (id==null || !id.matches("[a-z][a-z0-9_]{0,63}")) throw new IllegalArgumentException("invalid node id: "+id);
      Objects.requireNonNull(op,"op"); inputs=List.copyOf(inputs); parameters=List.copyOf(parameters);
      if (inputs.size()!=op.inputs || parameters.size()!=op.parameters) throw new IllegalArgumentException(id+": wrong arity for "+op);
      for (double p:parameters) if (!Double.isFinite(p)||Math.abs(p)>VALUE_LIMIT) throw new IllegalArgumentException(id+": invalid parameter");
      boolean needsNoise=op==Op.NOISE_2D||op==Op.NOISE_3D;
      if (needsNoise!=(noise!=null)) throw new IllegalArgumentException(id+": noise profile only allowed and required on noise nodes");
      if ((op==Op.CLAMP||op==Op.SMOOTHSTEP)&&parameters.get(0)>=parameters.get(1)) throw new IllegalArgumentException(id+": min must be below max");
      if (op==Op.WARP&&(parameters.getFirst()<0||parameters.getFirst()>65536)) throw new IllegalArgumentException(id+": warp amplitude must be 0..65536");
    }
    public static Node constant(String id,double value) { return new Node(id,Op.CONSTANT,List.of(),List.of(value),null); }
    public static Node noise(String id,NoiseSpec spec,boolean horizontal) { return new Node(id,horizontal?Op.NOISE_2D:Op.NOISE_3D,List.of(),List.of(),spec); }
    public static Node operation(String id,Op op,String... inputs) { return new Node(id,op,List.of(inputs),List.of(),null); }
    public static Node bounded(String id,Op op,String input,double min,double max) { return new Node(id,op,List.of(input),List.of(min,max),null); }
  }

  public record Definition(int version,String root,List<Node> nodes) {
    public Definition {
      if (version!=VERSION) throw new IllegalArgumentException("unsupported density graph version: "+version);
      Objects.requireNonNull(root,"root"); nodes=List.copyOf(nodes);
      if (nodes.isEmpty()||nodes.size()>MAX_NODES) throw new IllegalArgumentException("density graph needs 1..128 nodes");
    }
    public DensityGraph compile(long seed) { return new DensityGraph(this,seed); }
  }

  public record Bounds(double min,double max) {}
  private record Program(Node node,int[] inputs,SeededNoise3D noise,Bounds bounds,int depth,int cost) {}
  private final Definition definition;
  private final Program[] program;
  private final Map<String,Integer> ids;
  private final int root;
  private final int[] dependencies;
  private final boolean[] has3D;
  private final int[][] zeroFactors;

  private DensityGraph(Definition definition,long seed) {
    this.definition=definition; var index=new LinkedHashMap<String,Integer>();
    for (var n:definition.nodes()) if (index.putIfAbsent(n.id(),index.size())!=null) throw new IllegalArgumentException("duplicate node: "+n.id());
    Integer r=index.get(definition.root()); if(r==null) throw new IllegalArgumentException("missing root: "+definition.root());
    root=r; ids=Map.copyOf(index); program=new Program[index.size()];
    build(root,new byte[index.size()],seed);
    dependencies=new int[program.length];
    Arrays.fill(dependencies,-1);
    dependencies(root);
    has3D=new boolean[program.length];
    for(int i=0;i<program.length;i++)has3D[i]=contains3D(i);
    zeroFactors=new int[program.length][];
    for(int i=0;i<program.length;i++) {
      var factors=new LinkedHashSet<Integer>();
      if(program[i].node().op()==Op.ADD&&has3D[program[i].inputs()[1]])collectFactors(program[i].inputs()[1],factors);
      zeroFactors[i]=factors.stream().mapToInt(Integer::intValue).toArray();
    }
    for (int i=0;i<program.length;i++) if(program[i]==null) throw new IllegalArgumentException("unreachable node: "+definition.nodes().get(i).id());
  }

  private Program build(int at,byte[] state,long seed) {
    if(state[at]==1) throw new IllegalArgumentException("cycle at node: "+definition.nodes().get(at).id());
    if(program[at]!=null) return program[at];
    state[at]=1; Node n=definition.nodes().get(at); int[] inputs=new int[n.inputs().size()];
    int depth=1,cost=1;
    for(int i=0;i<inputs.length;i++) {
      Integer source=ids.get(n.inputs().get(i)); if(source==null) throw new IllegalArgumentException(n.id()+": missing input "+n.inputs().get(i));
      inputs[i]=source; Program p=build(source,state,seed); depth=Math.max(depth,1+p.depth()); cost=Math.min(MAX_COST+1,cost+p.cost());
    }
    if(depth>MAX_DEPTH||cost>MAX_COST) throw new IllegalArgumentException(n.id()+": density graph depth/work budget exceeded");
    Bounds b=bounds(n,inputs); if(!Double.isFinite(b.min())||!Double.isFinite(b.max())||Math.max(Math.abs(b.min()),Math.abs(b.max()))>VALUE_LIMIT)
      throw new IllegalArgumentException(n.id()+": density range exceeds finite budget");
    var p=new Program(n,inputs,n.noise()==null?null:new SeededNoise3D(seed,n.noise()),b,depth,cost);
    program[at]=p;state[at]=2;return p;
  }

  private Bounds bounds(Node n,int[] in) {
    Bounds a=in.length>0?program[in[0]].bounds():null,b=in.length>1?program[in[1]].bounds():null;
    return switch(n.op()) {
      case CONSTANT -> new Bounds(n.parameters().getFirst(),n.parameters().getFirst());
      case X,Y,Z -> new Bounds(-32_000_000,32_000_000);
      case NOISE_2D,NOISE_3D -> new Bounds(-1,1);
      case ADD -> new Bounds(a.min()+b.min(),a.max()+b.max());
      case MULTIPLY -> { double[] v={a.min()*b.min(),a.min()*b.max(),a.max()*b.min(),a.max()*b.max()}; yield new Bounds(Arrays.stream(v).min().orElseThrow(),Arrays.stream(v).max().orElseThrow()); }
      case MIN -> new Bounds(Math.min(a.min(),b.min()),Math.min(a.max(),b.max()));
      case MAX -> new Bounds(Math.max(a.min(),b.min()),Math.max(a.max(),b.max()));
      case ABS -> new Bounds(a.min()<=0&&a.max()>=0?0:Math.min(Math.abs(a.min()),Math.abs(a.max())),Math.max(Math.abs(a.min()),Math.abs(a.max())));
      case CLAMP -> new Bounds(clamp(a.min(),n.parameters().get(0),n.parameters().get(1)),clamp(a.max(),n.parameters().get(0),n.parameters().get(1)));
      case SMOOTHSTEP -> new Bounds(0,1);
      case LERP -> new Bounds(Math.min(a.min(),b.min()),Math.max(a.max(),b.max()));
      case WARP -> { for(int i=1;i<4;i++) {var w=program[in[i]].bounds();if(w.min() < -1||w.max()>1)throw new IllegalArgumentException(n.id()+": warp displacement fields must be bounded within [-1,1]");} yield a; }
    };
  }

  public Definition definition() { return definition; }
  public Bounds bounds() { return program[root].bounds(); }
  public int maximumEvaluationCost() { return program[root].cost(); }
  public int depth() { return program[root].depth(); }
  @Override public double sample(double x,double y,double z) { coordinates(x,y,z);return evaluate(root,x,y,z); }
  public double channel(String id,double x,double y,double z) {
    Integer node=ids.get(id);if(node==null)throw new IllegalArgumentException("unknown channel: "+id);
    coordinates(x,y,z);return evaluate(node,x,y,z);
  }
  private static void coordinates(double x,double y,double z) {
    coordinate(x);coordinate(y);coordinate(z);
  }
  private static void coordinate(double v) {if(!Double.isFinite(v)||Math.abs(v)>WORLD_LIMIT)throw new IllegalArgumentException("graph coordinate must be finite within +/-30000000");}
  private double evaluate(int at,double x,double y,double z) {
    Program p=program[at];var n=p.node();int[] i=p.inputs();
    return switch(n.op()) {
      case CONSTANT -> n.parameters().getFirst(); case X -> x; case Y -> y; case Z -> z;
      case NOISE_2D -> p.noise().sample(x,0,z); case NOISE_3D -> p.noise().sample(x,y,z);
      case ADD -> evaluate(i[0],x,y,z)+evaluate(i[1],x,y,z);
      case MULTIPLY -> evaluate(i[0],x,y,z)*evaluate(i[1],x,y,z);
      case MIN -> Math.min(evaluate(i[0],x,y,z),evaluate(i[1],x,y,z));
      case MAX -> Math.max(evaluate(i[0],x,y,z),evaluate(i[1],x,y,z));
      case ABS -> Math.abs(evaluate(i[0],x,y,z));
      case CLAMP -> clamp(evaluate(i[0],x,y,z),n.parameters().get(0),n.parameters().get(1));
      case SMOOTHSTEP -> {double t=clamp((evaluate(i[0],x,y,z)-n.parameters().get(0))/(n.parameters().get(1)-n.parameters().get(0)),0,1);yield t*t*(3-2*t);}
      case LERP -> {double a=evaluate(i[0],x,y,z),b=evaluate(i[1],x,y,z),t=clamp(evaluate(i[2],x,y,z),0,1);yield t==0?a:t==1?b:a+(b-a)*t;}
      case WARP -> {double amplitude=n.parameters().getFirst();yield evaluate(i[0],x+amplitude*evaluate(i[1],x,y,z),y+amplitude*evaluate(i[2],x,y,z),z+amplitude*evaluate(i[3],x,y,z));}
    };
  }
  private int dependencies(int at) {
    if(dependencies[at]>=0)return dependencies[at];
    Program p=program[at];int mask=switch(p.node().op()) {
      case CONSTANT -> 0; case X -> 1; case Y -> 2; case Z -> 4;
      case NOISE_2D -> 5; case NOISE_3D -> 7;
      default -> {int m=0;for(int input:p.inputs())m|=dependencies(input);yield m;}
    };
    return dependencies[at]=mask;
  }

  private boolean contains3D(int at) {
    if(program[at].node().op()==Op.NOISE_3D)return true;
    for(int child:program[at].inputs())if(contains3D(child))return true;
    return false;
  }
  private void collectFactors(int at,Set<Integer> factors) {
    if(factors.size()>=4)return;
    Program p=program[at];
    if(p.node().op()==Op.MULTIPLY) {
      collectFactors(p.inputs()[1],factors);collectFactors(p.inputs()[0],factors);
    } else if(!has3D[at]&&!(p.node().op()==Op.CONSTANT&&p.node().parameters().getFirst()!=0))factors.add(at);
  }

  /** Bounded exact memo storage owned by one job/thread; never shared by graph readers. */
  public Sampler newSampler() { return new Sampler(); }

  public final class Sampler implements DensityField {
    private final double[] values=new double[program.length];
    private final long[] xs=new long[program.length],ys=new long[program.length],zs=new long[program.length];
    private final boolean[] valid=new boolean[program.length];
    private final Thread owner=Thread.currentThread();

    private void owned() {
      if(owner!=Thread.currentThread())throw new IllegalStateException("Density sampler changed job thread");
    }
    @Override public double sample(double x,double y,double z) {
      owned();coordinates(x,y,z);return cached(root,x,y,z);
    }
    public double channel(String id,double x,double y,double z) {
      owned();Integer at=ids.get(id);if(at==null)throw new IllegalArgumentException("unknown channel: "+id);
      coordinates(x,y,z);return cached(at,x,y,z);
    }
    private double cached(int at,double x,double y,double z) {
      int mask=dependencies[at];
      long xb=Double.doubleToLongBits(x),yb=Double.doubleToLongBits(y),zb=Double.doubleToLongBits(z);
      if(valid[at]&&((mask&1)==0||xs[at]==xb)&&((mask&2)==0||ys[at]==yb)&&((mask&4)==0||zs[at]==zb))return values[at];
      double value=compute(at,x,y,z);
      xs[at]=xb;ys[at]=yb;zs[at]=zb;values[at]=value;valid[at]=true;
      return value;
    }
    private double compute(int at,double x,double y,double z) {
      Program p=program[at];var n=p.node();int[] i=p.inputs();
      return switch(n.op()) {
        case CONSTANT -> n.parameters().getFirst(); case X -> x; case Y -> y; case Z -> z;
        case NOISE_2D -> p.noise().sample(x,0,z); case NOISE_3D -> p.noise().sample(x,y,z);
        case ADD -> {
          double a=cached(i[0],x,y,z);boolean zero=false;
          // A finite nonzero addend is bit-identical after adding either sign of zero.
          // If it is zero, retain original evaluation to preserve signed-zero output.
          if(a!=0)for(int factor:zeroFactors[at])if(cached(factor,x,y,z)==0){zero=true;break;}
          yield zero?a:a+cached(i[1],x,y,z);
        }
        case MULTIPLY -> cached(i[0],x,y,z)*cached(i[1],x,y,z);
        case MIN -> Math.min(cached(i[0],x,y,z),cached(i[1],x,y,z));
        case MAX -> Math.max(cached(i[0],x,y,z),cached(i[1],x,y,z));
        case ABS -> Math.abs(cached(i[0],x,y,z));
        case CLAMP -> clamp(cached(i[0],x,y,z),n.parameters().get(0),n.parameters().get(1));
        case SMOOTHSTEP -> {double t=clamp((cached(i[0],x,y,z)-n.parameters().get(0))/(n.parameters().get(1)-n.parameters().get(0)),0,1);yield t*t*(3-2*t);}
        case LERP -> {double a=cached(i[0],x,y,z),b=cached(i[1],x,y,z),t=clamp(cached(i[2],x,y,z),0,1);yield t==0?a:t==1?b:a+(b-a)*t;}
        case WARP -> {double amplitude=n.parameters().getFirst();yield cached(i[0],x+amplitude*cached(i[1],x,y,z),y+amplitude*cached(i[2],x,y,z),z+amplitude*cached(i[3],x,y,z));}
      };
    }
  }

  private static double clamp(double value,double min,double max) { return Math.max(min,Math.min(max,value)); }
}
